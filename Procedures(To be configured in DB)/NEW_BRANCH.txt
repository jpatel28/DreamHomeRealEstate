create or replace PROCEDURE NEW_BRANCH (
    p_branchno IN VARCHAR2,
    p_street   IN VARCHAR2,
    p_city     IN VARCHAR2,
    p_postcode IN VARCHAR2
)
IS
BEGIN
    INSERT INTO dh_branch (branchno, street, city, postcode)
    VALUES (p_branchno, p_street, p_city, p_postcode);

    DBMS_OUTPUT.PUT_LINE('New branch ' || p_branchno || ' created successfully.');

    COMMIT;
EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        DBMS_OUTPUT.PUT_LINE('Branch number already exists: ' || p_branchno);
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error creating branch: ' || SQLERRM);
END NEW_BRANCH;
